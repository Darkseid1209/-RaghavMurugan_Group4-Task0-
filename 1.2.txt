import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
df = pd.read_csv("crime_train.csv")
df = df.drop(columns=['Unnamed: 0', 'Num', 'case_filed'])
df['weapon'] = df['weapon'].fillna('Unknown')
df['age'] = df['age'].fillna(df['age'].median())
df['police_department'] = df['police_department'].fillna(df['police_department'].mode()[0])
df['closed'] = df['closed'].map({'Yes': 1, 'No': 0})
categorical_cols = ['city', 'area', 'crime_description', 'sex', 'weapon', 'domain']
for col in categorical_cols:
    top_categories = df[col].value_counts().nlargest(15).index
    df[col] = df[col].where(df[col].isin(top_categories), 'Other')
df_encoded = pd.get_dummies(df, columns=categorical_cols, drop_first=True)
df_encoded = df_encoded.dropna().astype(float)
np.random.seed(42)
shuffled_indices = np.random.permutation(len(df_encoded))
test_set_size = int(len(df_encoded) * 0.2)

test_indices = shuffled_indices[:test_set_size]
train_indices = shuffled_indices[test_set_size:]

train_data = df_encoded.iloc[train_indices]
test_data = df_encoded.iloc[test_indices]

X_train = train_data.drop('closed', axis=1).values
y_train = train_data['closed'].values.reshape(-1, 1)

X_test = test_data.drop('closed', axis=1).values
y_test = test_data['closed'].values.reshape(-1, 1)
mean = np.mean(X_train, axis=0)
std = np.std(X_train, axis=0)
std[std == 0] = 1e-8 

X_train = (X_train - mean) / std
X_test = (X_test - mean) / std

class ScratchLogisticRegression:
    def __init__(self, learning_rate=0.5, epochs=1000):
        self.learning_rate = learning_rate
        self.epochs = epochs
        self.weights = None
        self.bias = None
        self.train_accuracies = []
        self.test_accuracies = []

    def sigmoid(self, z):
        z = np.clip(z, -250, 250)
        return 1 / (1 + np.exp(-z))

    def compute_accuracy(self, X, y):
        predictions = self.predict(X)
        return np.mean(predictions == y) * 100

    def fit(self, X_train, y_train, X_test, y_test):
        n_samples, n_features = X_train.shape
        self.weights = np.random.randn(n_features, 1) * 0.01 
        self.bias = 0

        for epoch in range(1, self.epochs + 1):
            linear_model = np.dot(X_train, self.weights) + self.bias
            y_predicted = self.sigmoid(linear_model)

            dw = (1 / n_samples) * np.dot(X_train.T, (y_predicted - y_train))
            db = (1 / n_samples) * np.sum(y_predicted - y_train)

            self.weights -= self.learning_rate * dw
            self.bias -= self.learning_rate * db
            if epoch % 100 == 0 or epoch == 1:
                train_acc = self.compute_accuracy(X_train, y_train)
                test_acc = self.compute_accuracy(X_test, y_test)
                self.train_accuracies.append(train_acc)
                self.test_accuracies.append(test_acc)
                print(f"Epoch {epoch:4d} | Train Accuracy: {train_acc:.2f}% | Test Accuracy: {test_acc:.2f}%")

    def predict(self, X):
        linear_model = np.dot(X, self.weights) + self.bias
        y_predicted = self.sigmoid(linear_model)
        return (y_predicted >= 0.5).astype(int)
model = ScratchLogisticRegression(learning_rate=0.5, epochs=1000)

print("Starting Training...")
model.fit(X_train, y_train, X_test, y_test)

plt.figure(figsize=(8, 5))
plt.plot([x * 100 for x in range(1, len(model.train_accuracies) + 1)], model.train_accuracies, label='Train Accuracy')
plt.plot([x * 100 for x in range(1, len(model.test_accuracies) + 1)], model.test_accuracies, label='Test Accuracy')
plt.title('Logistic Regression Accuracy Over Epochs')
plt.xlabel('Epochs')
plt.ylabel('Accuracy (%)')
plt.legend()
plt.grid(True)
plt.show()
